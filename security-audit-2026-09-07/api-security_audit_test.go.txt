func TestAuditInitialPasswordFullAccess(t *testing.T) {
 s,_:=testServer(t)
 s.cfg.Settings.MustChangePassword=true
 s.Mount(func(r chi.Router){r.Post("/api/audit-mutation",func(w http.ResponseWriter,r *http.Request){OK(w,nil)})})
 h:=s.Router()
 login:=apiRequest(h,"POST","/api/login",`{"username":"admin","password":"current-password"}`,"","192.0.2.1:1",nil)
 res:=apiRequest(h,"POST","/api/audit-mutation",`{}`,"","192.0.2.1:1",login.Result().Cookies()[0])
 if res.Code!=200 {t.Fatalf("status=%d",res.Code)}
 t.Log("mustChangePassword=true; protected mutation accepted with status 200")
}
func TestAuditCookiePortAndReplay(t *testing.T){
 s,h:=testServer(t)
 login:=apiRequest(h,"POST","/api/login",`{"username":"admin","password":"current-password"}`,"","192.0.2.1:1",nil)
 j,_:=cookiejar.New(nil)
 a,_:=url.Parse("https://router.local:16606/")
 b,_:=url.Parse("https://router.local:8443/")
 j.SetCookies(a,login.Result().Cookies())
 cookies:=j.Cookies(b)
 if len(cookies)!=1 {t.Fatal("cookie not shared")}
 res:=apiRequest(h,"GET","/api/me","","","198.51.100.22:1234",cookies[0])
 if res.Code!=200 || !s.tokens.Valid(cookies[0].Value){t.Fatal("replay failed")}
 t.Log("management cookie selected for HTTPS port 8443; replay from different IP accepted: 200")
}
type gatedBody struct {r *strings.Reader; ready chan struct{}; gate chan struct{}; once sync.Once}
func(b *gatedBody) Read(p []byte)(int,error){b.once.Do(func(){b.ready<-struct{}{};<-b.gate});return b.r.Read(p)}
func(b *gatedBody) Close()error{return nil}
func TestAuditConcurrentLoginLimit(t *testing.T){
 _,h:=testServer(t)
 const n=12
 ready:=make(chan struct{},n);gate:=make(chan struct{});results:=make(chan int,n)
 for i:=0;i<n;i++{go func(){r:=httptest.NewRequest("POST","https://router.local/api/login",nil);r.RemoteAddr="192.0.2.44:1";r.Body=&gatedBody{r:strings.NewReader(`{"username":"admin","password":"wrong"}`),ready:ready,gate:gate};w:=httptest.NewRecorder();h.ServeHTTP(w,r);results<-w.Code}()}
 for i:=0;i<n;i++{<-ready};close(gate)
 counts:=map[int]int{};for i:=0;i<n;i++{counts[<-results]++}
 if counts[403]!=n {t.Fatalf("unexpected=%v",counts)}
 t.Logf("same-IP simultaneous attempts=%d, results=%v; documented limit=5",n,counts)
}
