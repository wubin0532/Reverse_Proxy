package webproxy
import("testing";"net/http";"net/http/httptest";"andey-proxy/internal/config")
func TestAuditManagementCookieForwarding(t *testing.T){
 received:=""
 b:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){received=r.Header.Get("Cookie");w.WriteHeader(200)}));defer b.Close()
 h:=securityTestReverseHandler(t,config.SubRule{Backends:[]string{b.URL}})
 r:=httptest.NewRequest("GET","https://router.local:8443/",nil)
 r.Header.Set("Cookie","andey-proxy_token=SYNTHETIC-AUDIT-TOKEN; app=ok")
 h.ServeHTTP(httptest.NewRecorder(),r)
 if received!="andey-proxy_token=SYNTHETIC-AUDIT-TOKEN; app=ok"{t.Fatal("not forwarded")}
 t.Log("backend received management cookie unchanged (synthetic token only)")
}
